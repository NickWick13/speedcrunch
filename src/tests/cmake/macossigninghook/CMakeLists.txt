cmake_minimum_required(VERSION 3.19)
project(testmacossigninghook NONE)
set(source_root "${CMAKE_CURRENT_LIST_DIR}/../../..")

set(CPACK_GENERATOR DragNDrop)
set(CPACK_TEMPORARY_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/missing-stage")
set(CPACK_SPEEDCRUNCH_MACOS_APP_BUNDLE SpeedCrunch.app)
set(ENV{SPEEDCRUNCH_MACOS_SIGN_IDENTITY} "Developer ID Application: Fixture User (ABCDEFGHIJ)")
if(TEST_PHASE STREQUAL "UNSET_PRE")
    unset(ENV{SPEEDCRUNCH_MACOS_SIGN_IDENTITY})
elseif(TEST_PHASE STREQUAL "UNSET_POST")
    unset(ENV{SPEEDCRUNCH_MACOS_SIGN_IDENTITY})
    set(CPACK_PACKAGE_FILES "${CMAKE_CURRENT_BINARY_DIR}/missing.dmg")
elseif(TEST_PHASE STREQUAL "EMPTY")
    set(ENV{SPEEDCRUNCH_MACOS_SIGN_IDENTITY} "   ")
elseif(TEST_PHASE STREQUAL "SOURCE")
    set(CPACK_SOURCE_INSTALLED_DIRECTORIES "${source_root};/")
elseif(TEST_PHASE STREQUAL "ZIP_POST")
    set(CPACK_GENERATOR ZIP)
    set(CPACK_PACKAGE_FILES "${CMAKE_CURRENT_BINARY_DIR}/missing.zip")
elseif(NOT TEST_PHASE STREQUAL "MISSING_APP")
    message(FATAL_ERROR "Unknown hook phase: ${TEST_PHASE}")
endif()

include("${source_root}/cmake/CPackMacOSSign.cmake")
