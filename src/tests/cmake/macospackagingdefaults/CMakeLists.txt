cmake_minimum_required(VERSION 3.19)

set(source_root "${CMAKE_CURRENT_LIST_DIR}/../../..")
if(NOT TEST_VARIANT STREQUAL "DEFAULT")
    set("BUILD_VARIANT_${TEST_VARIANT}" ON CACHE BOOL "Build variant under test" FORCE)
endif()
include("${source_root}/cmake/BuildVariants.cmake")
include("${source_root}/cmake/MacOSDeploymentTarget.cmake")
project(testmacospackagingdefaults NONE)

# Old cache entries must not turn signing on after the environment is cleared.
set(SPEEDCRUNCH_MACOS_SIGN_IDENTITY "Old cached identity" CACHE STRING "Unused old setting" FORCE)
set(SPEEDCRUNCH_MACOS_RELEASE ON CACHE BOOL "Unused old setting" FORCE)
set(ENV{SPEEDCRUNCH_MACOS_SIGN_IDENTITY} "")
set(APPLICATION_NAME SpeedCrunch)
set(SPEEDCRUNCH_VERSION "1.2.3")
list(PREPEND CMAKE_MODULE_PATH "${source_root}/cmake")
include("${source_root}/SpeedCrunchCPack.cmake")

if(TARGET macos-release)
    message(FATAL_ERROR "Packaging must not introduce a separate release target")
endif()
foreach(setting IN ITEMS SPEEDCRUNCH_MACOS_NOTARY_PROFILE SPEEDCRUNCH_MACOS_RELEASE_DIRECTORY)
    if(DEFINED ${setting})
        message(FATAL_ERROR "Packaging must not configure notarization: ${setting}")
    endif()
endforeach()

include("${CMAKE_CURRENT_BINARY_DIR}/CPackConfig.cmake")
if(TEST_VARIANT STREQUAL "OFFICIAL_PORTABLE_RELEASE")
    set(expected_generator ZIP)
    if(NOT PORTABLE_SPEEDCRUNCH)
        message(FATAL_ERROR "The portable build must keep portable settings storage enabled")
    endif()
else()
    set(expected_generator DragNDrop)
endif()
if(NOT CPACK_GENERATOR STREQUAL expected_generator OR NOT CPACK_DMG_FORMAT STREQUAL "UDBZ")
    message(FATAL_ERROR "The existing generator and DMG format must stay unchanged")
endif()
if(NOT CPACK_PACKAGE_FILE_NAME STREQUAL "SpeedCrunch-1.2.3-macos-arm64")
    message(FATAL_ERROR "Keep the existing package naming")
endif()
if(NOT CPACK_PRE_BUILD_SCRIPTS STREQUAL CPACK_POST_BUILD_SCRIPTS)
    message(FATAL_ERROR "Use the signing-only hook for both packaging phases")
endif()

foreach(hook IN LISTS CPACK_PRE_BUILD_SCRIPTS)
    include("${hook}")
endforeach()
set(CPACK_PACKAGE_FILES "${CMAKE_CURRENT_BINARY_DIR}/does-not-exist.dmg")
foreach(hook IN LISTS CPACK_POST_BUILD_SCRIPTS)
    include("${hook}")
endforeach()
if(EXISTS "${CPACK_PACKAGE_FILES}")
    message(FATAL_ERROR "An unset identity must leave artifacts alone")
endif()

foreach(helper IN ITEMS CPackMacOSSign.cmake MacOSSigning.cmake MacOSPackageSigning.cmake)
    file(READ "${source_root}/cmake/${helper}" code)
    if(code MATCHES "notarytool|stapler")
        message(FATAL_ERROR "Automatic signing must not run notarization or stapling: ${helper}")
    endif()
endforeach()
