cmake_minimum_required(VERSION 3.19)
project(testmacossigning NONE)
include("${CMAKE_CURRENT_LIST_DIR}/../../../cmake/MacOSSigning.cmake")

set(app "${CMAKE_CURRENT_BINARY_DIR}/Fixture With Spaces.app")
file(REMOVE_RECURSE "${app}" "${CMAKE_CURRENT_BINARY_DIR}/copy")
set(framework "${app}/Contents/Frameworks/Example.framework")
set(helper "${app}/Contents/Helpers/Helper.app")
file(MAKE_DIRECTORY "${framework}/Versions/A" "${helper}/Contents/MacOS"
    "${app}/Contents/MacOS" "${app}/Contents/Resources")
string(ASCII 207 250 237 254 macho)
foreach(binary IN ITEMS "${framework}/Versions/A/Example"
        "${helper}/Contents/MacOS/Helper" "${app}/Contents/MacOS/Main"
        "${app}/Contents/MacOS/tool-without-extension")
    file(WRITE "${binary}" "${macho}fixture")
endforeach()
file(WRITE "${app}/Contents/Resources/data.dylib" "This is a resource, not Mach-O code.")
file(CREATE_LINK A "${framework}/Versions/Current" SYMBOLIC)
file(CREATE_LINK Versions/Current/Example "${framework}/Example" SYMBOLIC)
speedcrunch_macos_signing_order("${app}" code)
list(LENGTH code count)
if(NOT count EQUAL 7)
    message(FATAL_ERROR "Expected four binaries and three bundles, got: ${code}")
endif()
foreach(pair IN ITEMS "${framework}/Versions/A/Example|${framework}"
        "${helper}/Contents/MacOS/Helper|${helper}" "${helper}|${app}"
        "${framework}|${app}" "${app}/Contents/MacOS/Main|${app}")
    string(REPLACE "|" ";" pair "${pair}")
    list(GET pair 0 child)
    list(GET pair 1 parent)
    list(FIND code "${child}" child_index)
    list(FIND code "${parent}" parent_index)
    if(child_index LESS 0 OR parent_index LESS 0 OR NOT child_index LESS parent_index)
        message(FATAL_ERROR "Nested code must be signed before its bundle: ${pair}")
    endif()
endforeach()

set(development "AAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAA")
set(developer_id "ABCDEF0123456789ABCDEF0123456789ABCDEF01")
set(name "Developer ID Application: Fixture User (ABCDEFGHIJ)")
set(identities "  1) ${development} \"Apple Development: Fixture User\"\n  2) ${developer_id} \"${name}\"\n")
string(TOLOWER "${developer_id}" lowercase)
foreach(requested IN ITEMS "${name}" "${developer_id}" "${lowercase}")
    speedcrunch_macos_select_identity("${identities}" "${requested}" actual)
    if(NOT actual STREQUAL developer_id)
        message(FATAL_ERROR "Cannot select the Developer ID identity: ${requested}")
    endif()
endforeach()
foreach(requested IN ITEMS "${development}" "Apple Development: Fixture User" "Fixture User" "-")
    speedcrunch_macos_select_identity("${identities}" "${requested}" actual)
    if(actual)
        message(FATAL_ERROR "Do not substitute another certificate or ad hoc signing: ${requested}")
    endif()
endforeach()
